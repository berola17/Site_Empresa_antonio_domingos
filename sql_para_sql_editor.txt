-- 1) Tabela de funcionarios
create table if not exists public.funcionarios (
  id           bigint generated by default as identity primary key,
  nome         text        not null,
  email        text        not null,
  cargo        text,
  departamento text,
  foto_url     text,
  created_at   timestamptz not null default now()
);

-- 2) RLS habilitado
alter table public.funcionarios enable row level security;

-- 3) Policies de leitura e insercao (cria apenas se nao existir)
do $$
begin
  if not exists (
    select 1 from pg_policies
    where schemaname = 'public'
      and tablename  = 'funcionarios'
      and policyname = 'funcionarios_select_publico'
  ) then
    create policy "funcionarios_select_publico"
      on public.funcionarios for select
      to anon, authenticated
      using (true);
  end if;

  if not exists (
    select 1 from pg_policies
    where schemaname = 'public'
      and tablename  = 'funcionarios'
      and policyname = 'funcionarios_insert_publico'
  ) then
    create policy "funcionarios_insert_publico"
      on public.funcionarios for insert
      to anon, authenticated
      with check (true);
  end if;
end $$;

-- 4) Bucket publico para as fotos de perfil
insert into storage.buckets (id, name, public)
values ('fotos-funcionarios', 'fotos-funcionarios', true)
on conflict (id) do update set public = true;

-- 5) Policies do bucket (cria apenas se nao existir)
do $$
begin
  if not exists (
    select 1 from pg_policies
    where schemaname = 'storage'
      and tablename  = 'objects'
      and policyname = 'fotos_upload_publico'
  ) then
    create policy "fotos_upload_publico"
      on storage.objects for insert
      to anon, authenticated
      with check (bucket_id = 'fotos-funcionarios');
  end if;

  if not exists (
    select 1 from pg_policies
    where schemaname = 'storage'
      and tablename  = 'objects'
      and policyname = 'fotos_leitura_publica'
  ) then
    create policy "fotos_leitura_publica"
      on storage.objects for select
      to anon, authenticated
      using (bucket_id = 'fotos-funcionarios');
  end if;
end $$;

-- 6) Indice para a listagem ordenada por id desc
create index if not exists funcionarios_id_idx on public.funcionarios (id desc);
